class Rational {
  final int numerator;
  final int denominator;

  const Rational._(this.numerator, this.denominator);

  factory Rational.parse(String input) {
    final text = input.trim();
    if (text.isEmpty) return Rational.zero;
    if (text.contains('/')) {
      final parts = text.split('/');
      if (parts.length != 2) throw FormatException('Invalid fraction: $input');
      final numerator = int.tryParse(parts[0].trim());
      final denominator = int.tryParse(parts[1].trim());
      if (numerator == null || denominator == null || denominator == 0) {
        throw FormatException('Invalid fraction: $input');
      }
      return Rational(numerator, denominator);
    }
    if (text.contains('.')) {
      final negative = text.startsWith('-');
      final unsigned = text.replaceFirst('-', '');
      final parts = unsigned.split('.');
      if (parts.length != 2) {
        throw FormatException('Invalid decimal: $input');
      }
      final whole = int.tryParse(parts.first);
      final digits = parts[1].replaceAll(RegExp(r'0+$'), '');
      final fraction = digits.isEmpty ? 0 : int.tryParse(digits);
      if (whole == null || fraction == null) {
        throw FormatException('Invalid decimal: $input');
      }
      final scale = _tenPow(digits.length);
      var result = whole * scale + fraction;
      if (negative) result = -result;
      return Rational(result, scale);
    }
    final value = int.tryParse(text);
    if (value == null) throw FormatException('Invalid number: $input');
    return Rational(value);
  }

  factory Rational(int numerator, [int denominator = 1]) {
    if (denominator == 0) throw ArgumentError('Denominator cannot be zero');
    var n = numerator;
    var d = denominator;
    if (d < 0) {
      n = -n;
      d = -d;
    }
    final divisor = _gcd(n.abs(), d);
    return Rational._(n ~/ divisor, d ~/ divisor);
  }

  static const Rational zero = Rational._(0, 1);
  static const Rational one = Rational._(1, 1);

  Rational operator +(Rational other) => Rational(
    numerator * other.denominator + other.numerator * denominator,
    denominator * other.denominator,
  );
  Rational operator -(Rational other) => Rational(
    numerator * other.denominator - other.numerator * denominator,
    denominator * other.denominator,
  );
  Rational operator *(Rational other) =>
      Rational(numerator * other.numerator, denominator * other.denominator);
  Rational operator /(Rational other) {
    if (other.numerator == 0) throw ArgumentError('Division by zero');
    return Rational(
      numerator * other.denominator,
      denominator * other.numerator,
    );
  }

  Rational operator -() => Rational(-numerator, denominator);

  bool get isZero => numerator == 0;
  bool get isNegative => numerator < 0;

  @override
  bool operator ==(Object other) =>
      other is Rational &&
      numerator == other.numerator &&
      denominator == other.denominator;
  @override
  int get hashCode => Object.hash(numerator, denominator);

  @override
  String toString() =>
      denominator == 1 ? '$numerator' : '$numerator/$denominator';

  static int _gcd(int a, int b) {
    while (b != 0) {
      final r = a % b;
      a = b;
      b = r;
    }
    return a == 0 ? 1 : a;
  }

  static int _tenPow(int exponent) {
    var value = 1;
    for (var i = 0; i < exponent; i++) {
      value *= 10;
    }
    return value;
  }
}

class GaussianService extends LinearSolverService {
  @override
  ({SolutionResult solution, List<SolverStep> steps}) solve(
    AugmentedMatrix matrix,
  ) {
    final work = matrix.copy();
    final rows = work.length;
    final cols = rows == 0 ? 0 : work.first.length - 1;
    final elimination = forwardEliminate(work, eliminateAbove: false);
    final steps = elimination.steps;
    final pivotRow = elimination.pivotRow;

    final rank = _coefficientRank(work);
    final augmentedRank = _augmentedRank(work);
    late final SolutionResult solution;
    if (rank < augmentedRank) {
      solution = const SolutionResult(
        type: SolutionType.noSolution,
        summary: 'No solution',
      );
    } else if (rank < cols) {
      solution = _parametricSolution(work);
    } else {
      final values = List<String>.filled(cols, '0');
      for (var r = cols - 1; r >= 0; r--) {
        var pivotCol = -1;
        for (var c = 0; c < cols; c++) {
          if (!work[r][c].isZero) {
            pivotCol = c;
            break;
          }
        }
        if (pivotCol < 0) continue;
        var value = work[r][cols];
        for (var c = pivotCol + 1; c < cols; c++) {
          value = value - work[r][c] * Rational.parse(values[c]);
        }
        values[pivotCol] = (value / work[r][pivotCol]).toString();
      }
      solution = SolutionResult(
        type: SolutionType.unique,
        values: values,
        summary: 'Unique solution',
      );
      steps.add(
        SolverStep(
          matrix: work.copy(),
          operationType: 'BACK-SUB',
          description:
              'Use back-substitution to solve from the last row upward',
          formula: values
              .asMap()
              .entries
              .map((entry) => 'x${subscript(entry.key + 1)} = ${entry.value}')
              .join(', '),
          pivotRow: cols - 1,
          pivotCol: cols - 1,
        ),
      );
    }
    steps.add(
      SolverStep(
        matrix: work.copy(),
        operationType: 'COMPLETE',
        description: solution.summary,
        formula: 'Complete',
        pivotRow: pivotRow > 0 ? pivotRow - 1 : null,
      ),
    );
    return (solution: solution, steps: steps);
  }

  int _coefficientRank(AugmentedMatrix matrix) {
    if (matrix.isEmpty) return 0;
    var rank = 0;
    final cols = matrix.first.length - 1;
    for (var r = 0; r < matrix.length; r++) {
      for (var c = 0; c < cols; c++) {
        if (!matrix[r][c].isZero) {
          rank++;
          break;
        }
      }
    }
    return rank;
  }

  int _augmentedRank(AugmentedMatrix matrix) {
    if (matrix.isEmpty) return 0;
    var rank = 0;
    for (var r = 0; r < matrix.length; r++) {
      if (matrix[r].any((value) => !value.isZero)) rank++;
    }
    return rank;
  }

  SolutionResult _parametricSolution(AugmentedMatrix matrix) {
    final cols = matrix.first.length - 1;
    final pivotColumns = <int>{};
    for (var r = 0; r < matrix.length; r++) {
      for (var c = 0; c < cols; c++) {
        if (!matrix[r][c].isZero) {
          pivotColumns.add(c);
          break;
        }
      }
    }
    final expressions = <int, ParametricExpression>{};
    for (var c = 0; c < cols; c++) {
      if (!pivotColumns.contains(c)) {
        expressions[c] = ParametricExpression.free(c);
      }
    }
    for (var r = matrix.length - 1; r >= 0; r--) {
      var pivotCol = -1;
      for (var c = 0; c < cols; c++) {
        if (!matrix[r][c].isZero) {
          pivotCol = c;
          break;
        }
      }
      if (pivotCol < 0) continue;

      final pivot = matrix[r][pivotCol];
      var expression = ParametricExpression(matrix[r][cols] / pivot);
      for (var c = pivotCol + 1; c < cols; c++) {
        if (!matrix[r][c].isZero) {
          final coefficient = -matrix[r][c] / pivot;
          expression = expression.add(expressions[c]!.scaled(coefficient));
        }
      }
      expressions[pivotCol] = expression;
    }
    return SolutionResult(
      type: SolutionType.infinite,
      parametricExpressions: {
        for (final entry in expressions.entries)
          entry.key: pivotColumns.contains(entry.key)
              ? formatParametricExpression(entry.value)
              : 'free',
      },
      summary: 'Infinitely many solutions',
    );
  }
}

abstract class LinearSolverService {
  ({SolutionResult solution, List<SolverStep> steps}) solve(
    AugmentedMatrix matrix,
  );

  String rowLabel(int index) => 'R${subscript(index + 1)}';

  String formatFactor(Rational value) => value.denominator == 1
      ? value.numerator.toString()
      : '${value.numerator}/${value.denominator}';

  String parenthesize(Rational value) => value.denominator == 1
      ? value.numerator.toString()
      : '(${value.numerator}/${value.denominator})';

  String subscript(int value) =>
      value >= 1 && value <= AppConstants.variableSubscripts.length
      ? AppConstants.variableSubscripts[value - 1]
      : value.toString().split('').map((digit) {
          const digits = ['₀', '₁', '₂', '₃', '₄', '₅', '₆', '₇', '₈', '₉'];
          return digits[int.parse(digit)];
        }).join();

  String formatParametricExpression(ParametricExpression expression) {
    var result = '';
    if (!expression.constant.isZero) result = expression.constant.toString();

    final sortedCoefficients = expression.coefficients.entries.toList()
      ..sort((a, b) => a.key.compareTo(b.key));
    for (final entry in sortedCoefficients) {
      final c = entry.key;
      final coefficient = entry.value;
      if (coefficient.isZero) continue;

      final variable = 'x${subscript(c + 1)}';
      final magnitude = coefficient.isNegative ? -coefficient : coefficient;
      final coefficientText = magnitude == Rational.one ? '' : '$magnitude';
      final term = '$coefficientText$variable';

      if (coefficient.isNegative) {
        result = result.isEmpty ? '- $term' : '$result - $term';
      } else {
        result = result.isEmpty ? term : '$result + $term';
      }
    }

    return result.isEmpty ? '0' : result;
  }

  ({List<SolverStep> steps, List<int> pivotColumns, int pivotRow})
  forwardEliminate(AugmentedMatrix work, {required bool eliminateAbove}) {
    final rows = work.length;
    final cols = rows == 0 ? 0 : work.first.length - 1;
    final pivotColumns = <int>[];
    final steps = <SolverStep>[
      SolverStep(
        matrix: work.copy(),
        operationType: 'INITIAL',
        description: 'Begin with the entered augmented matrix',
        formula: '[A | b]',
      ),
    ];
    var pivotRow = 0;

    for (var col = 0; col < cols && pivotRow < rows; col++) {
      var selected = pivotRow;
      while (selected < rows && work[selected][col].isZero) {
        selected++;
      }
      if (selected == rows) continue;
      if (selected != pivotRow) {
        work.swapRows(selected, pivotRow);
        steps.add(
          SolverStep(
            matrix: work.copy(),
            operationType: 'SWAP',
            description: eliminateAbove
                ? 'Swap ${rowLabel(pivotRow)} and ${rowLabel(selected)} to bring a non-zero pivot into position'
                : 'Swap R${pivotRow + 1} and R${selected + 1} to bring a non-zero pivot into position',
            formula: '${rowLabel(pivotRow)} ↔ ${rowLabel(selected)}',
            pivotRow: pivotRow,
            pivotCol: col,
            sourceRow: selected,
            targetRow: pivotRow,
          ),
        );
      }

      final pivot = work[pivotRow][col];
      if (pivot != Rational.one) {
        for (var c = col; c <= cols; c++) {
          work[pivotRow][c] = work[pivotRow][c] / pivot;
        }
        steps.add(
          SolverStep(
            matrix: work.copy(),
            operationType: 'SCALE',
            description: eliminateAbove
                ? 'Scale ${rowLabel(pivotRow)} so the leading entry becomes 1'
                : 'Scale ${rowLabel(pivotRow)} by ${formatFactor(Rational.one / pivot)} so its leading entry becomes 1',
            formula:
                '${rowLabel(pivotRow)} ← ${parenthesize(Rational.one / pivot)}·${rowLabel(pivotRow)}',
            pivotRow: pivotRow,
            pivotCol: col,
            targetRow: pivotRow,
          ),
        );
      }

      pivotColumns.add(col);
      final startRow = eliminateAbove ? 0 : pivotRow + 1;
      final endRow = eliminateAbove ? rows : rows;
      for (var r = startRow; r < endRow; r++) {
        if (eliminateAbove && r == pivotRow) continue;
        if (!eliminateAbove && r <= pivotRow) continue;
        final factor = work[r][col];
        if (factor.isZero) continue;
        for (var c = col; c <= cols; c++) {
          work[r][c] = work[r][c] - factor * work[pivotRow][c];
        }
        steps.add(
          SolverStep(
            matrix: work.copy(),
            operationType: 'ELIMINATE',
            description:
                'Eliminate x${subscript(col + 1)} in ${rowLabel(r)} using the pivot in ${rowLabel(pivotRow)}',
            formula:
                '${rowLabel(r)} ← ${rowLabel(r)} - ${formatFactor(factor)}·${rowLabel(pivotRow)}',
            pivotRow: pivotRow,
            pivotCol: col,
            sourceRow: pivotRow,
            targetRow: r,
          ),
        );
      }
      pivotRow++;
    }

    return (steps: steps, pivotColumns: pivotColumns, pivotRow: pivotRow);
  }
}

class ParametricExpression {
  final Rational constant;
  final Map<int, Rational> coefficients;

  ParametricExpression(this.constant, [Map<int, Rational>? coefficients])
    : coefficients = coefficients ?? <int, Rational>{};

  ParametricExpression.free(int variable)
    : constant = Rational.zero,
      coefficients = <int, Rational>{variable: Rational.one};

  ParametricExpression scaled(Rational factor) {
    final scaledCoefficients = <int, Rational>{};
    for (final entry in coefficients.entries) {
      scaledCoefficients[entry.key] = entry.value * factor;
    }
    return ParametricExpression(constant * factor, scaledCoefficients);
  }

  ParametricExpression add(ParametricExpression other) {
    final combined = Map<int, Rational>.from(coefficients);
    for (final entry in other.coefficients.entries) {
      combined[entry.key] =
          (combined[entry.key] ?? Rational.zero) + entry.value;
    }
    return ParametricExpression(constant + other.constant, combined);
  }
}

class SolverViewModel extends ChangeNotifier {
  int rows = AppConstants.defaultRows;
  int cols = AppConstants.defaultCols;
  bool isGaussJordan = true;
  late List<List<String>> grid;
  int gridVersion = 0;
  String? errorMessage;
  PresetType? _selectedPreset;
  PresetType? get selectedPreset => _selectedPreset;

  List<SolverStep> steps = [];
  SolutionResult? solutionResult;
  int currentStepIndex = 0;
  bool isPlaying = false;
  double playbackSpeed = AppConstants.defaultPlaybackSpeed;
  Timer? _timer;

  SolverViewModel() {
    grid = List.generate(rows, (_) => List.generate(cols + 1, (_) => ''));
  }

  SolverStep? get currentStep => steps.isEmpty ? null : steps[currentStepIndex];
  bool get hasSolution => solutionResult != null;

  void resizeGrid(int newRows, int newCols) {
    if (newRows < AppConstants.minDimension ||
        newRows > AppConstants.maxDimension ||
        newCols < AppConstants.minDimension ||
        newCols > AppConstants.maxDimension ||
        (newRows == rows && newCols == cols)) {
      return;
    }
    final updated = List.generate(newRows, (r) {
      return List.generate(newCols + 1, (c) {
        if (r < rows && c < cols) return grid[r][c];
        if (r < rows && c == newCols) return grid[r][cols];
        return '';
      });
    });
    rows = newRows;
    cols = newCols;
    grid = updated;
    gridVersion++;
    _selectedPreset = null;
    _resetSolution();
  }

  void updateCell(int r, int c, String val) {
    grid[r][c] = val.trim();
    _selectedPreset = null;
    notifyListeners();
  }

  void clearGrid() {
    grid = List.generate(rows, (_) => List.generate(cols + 1, (_) => ''));
    gridVersion++;
    _selectedPreset = null;
    _resetSolution();
  }

  void applyPreset(PresetType type) {
    rows = AppConstants.defaultRows;
    cols = AppConstants.defaultCols;
    grid = MatrixPresets.forType(type);
    gridVersion++;
    _selectedPreset = type;
    _resetSolution();
  }

  void setMethod(bool isGaussJordanSelected) {
    if (isGaussJordan == isGaussJordanSelected) return;
    isGaussJordan = isGaussJordanSelected;
    _resetSolution();
  }

  void solve() {
    stopPlayback();
    errorMessage = null;
    try {
      final matrix = List<List<Rational>>.generate(rows, (r) {
        return List<Rational>.generate(cols + 1, (c) {
          final text = grid[r][c].trim();
          if (text.isEmpty) return Rational.zero;
          return Rational.parse(text);
        });
      });
      final result = isGaussJordan
          ? GaussJordanService().solve(matrix)
          : GaussianService().solve(matrix);
      steps = result.steps;
      solutionResult = result.solution;
      currentStepIndex = 0;
      if (steps.length > 1) {
        isPlaying = true;
        _restartTimer();
      }
    } on FormatException catch (error) {
      steps = [];
      solutionResult = null;
      currentStepIndex = 0;
      errorMessage = error.message;
    } on ArgumentError catch (error) {
      steps = [];
      solutionResult = null;
      currentStepIndex = 0;
      errorMessage = error.message?.toString() ?? 'Invalid matrix value';
    }
    notifyListeners();
  }

  void nextStep() {
    if (steps.isEmpty || currentStepIndex >= steps.length - 1) {
      if (isPlaying) stopPlayback();
      return;
    }
    currentStepIndex++;
    if (currentStepIndex >= steps.length - 1 && isPlaying) {
      stopPlayback();
    }
    notifyListeners();
  }

  void previousStep() {
    if (steps.isEmpty || currentStepIndex == 0) return;
    currentStepIndex--;
    notifyListeners();
  }

  void goToStep(int index) {
    if (steps.isEmpty) return;
    currentStepIndex = index.clamp(0, steps.length - 1);
    notifyListeners();
  }

  void togglePlayPause() {
    if (steps.isEmpty) return;
    if (isPlaying) {
      stopPlayback();
      notifyListeners();
    } else {
      if (currentStepIndex >= steps.length - 1) currentStepIndex = 0;
      isPlaying = true;
      notifyListeners();
      _restartTimer();
    }
  }

  void stopPlayback() {
    _timer?.cancel();
    _timer = null;
    if (isPlaying) {
      isPlaying = false;
    }
  }

  void cyclePlaybackSpeed() {
    final speeds = AppConstants.playbackSpeeds;
    final index = speeds.indexOf(playbackSpeed);
    playbackSpeed = speeds[(index + 1) < speeds.length ? index + 1 : 0];
    if (isPlaying) _restartTimer();
    notifyListeners();
  }

  void _restartTimer() {
    _timer?.cancel();
    final interval = Duration(
      milliseconds: (AppConstants.baseStepIntervalMs / playbackSpeed).round(),
    );
    _timer = Timer.periodic(interval, (_) {
      if (steps.isEmpty || currentStepIndex >= steps.length - 1) {
        stopPlayback();
        notifyListeners();
        return;
      }
      currentStepIndex++;
      if (currentStepIndex >= steps.length - 1) {
        stopPlayback();
      }
      notifyListeners();
    });
  }

  void _resetSolution() {
    stopPlayback();
    steps = [];
    solutionResult = null;
    currentStepIndex = 0;
    errorMessage = null;
    notifyListeners();
  }

  @override
  void dispose() {
    _timer?.cancel();
    super.dispose();
  }
}

  Widget _buildCell(
    int r,
    int c,
    AugmentedMatrix? matrix, {
    bool isConstant = false,
  }) {
    final step = viewModel.currentStep;
    final isPivot = r == step?.pivotRow && c == step?.pivotCol;
    final isTargetCell =
        r == step?.targetRow &&
        c == step?.pivotCol &&
        step?.targetRow != null &&
        step?.pivotCol != null;
    String text;
    if (matrix == null) {
      text = viewModel.grid[r][c].trim().isEmpty
          ? '0'
          : viewModel.grid[r][c].trim();
    } else {
      text = matrix[r][c].toString();
    }

    var fill = isConstant ? AppTheme.subtleFill : AppTheme.surfaceWhite;
    if (isTargetCell) {
      fill = AppTheme.brandy;
    }
    if (isPivot) fill = AppTheme.jasmine;

    var borderColor = isTargetCell ? AppTheme.brandy : AppTheme.borderSubtle;
    var borderWidth = isTargetCell ? 1.5 : 1.0;
    if (isPivot) {
      borderColor = AppTheme.inkBlack;
      borderWidth = 2.0;
    }

    return AnimatedContainer(
      duration: AppConstants.stepTransitionDuration,
      width: AppConstants.stageCellWidth,
      height: AppConstants.stageCellHeight,
      margin: const EdgeInsets.symmetric(horizontal: 4),
      decoration: BoxDecoration(
        color: fill,
        borderRadius: BorderRadius.circular(8),
        border: Border.all(color: borderColor, width: borderWidth),
      ),
      child: Center(
        child: Text(
          text,
          style: isPivot
              ? AppTheme.matrixNumberStyle.copyWith(
                  fontWeight: FontWeight.w800,
                  color: AppTheme.inkBlack,
                )
              : AppTheme.matrixNumberStyle,
        ),
      ),
    );
  }

  Widget _buildSolutionSummary() {
    final result = viewModel.solutionResult;
    if (result == null) return const SizedBox.shrink();
    final lines = <String>[];
    if (result.type == SolutionType.unique) {
      for (var i = 0; i < result.values.length; i++) {
        lines.add(
          'x${AppConstants.variableSubscripts[i]} = ${result.values[i]}',
        );
      }
    } else if (result.type == SolutionType.infinite) {
      for (var i = 0; i < viewModel.cols; i++) {
        final variable = 'x${AppConstants.variableSubscripts[i]}';
        final value = result.parametricExpressions[i];
        lines.add(
          value == 'free'
              ? '$variable is free'
              : '$variable = ${value ?? 'free'}',
        );
      }
    } else {
      lines.add('No values satisfy all equations simultaneously.');
    }
    return Container(
      margin: const EdgeInsets.only(top: 12),
      padding: const EdgeInsets.symmetric(horizontal: 16, vertical: 10),
      decoration: BoxDecoration(
        color: AppTheme.canvasBg,
        borderRadius: BorderRadius.circular(10),
        border: Border.all(color: AppTheme.borderStrong),
      ),
      child: Column(
        mainAxisSize: MainAxisSize.min,
        children: [
          Text(
            result.summary,
            style: const TextStyle(
              fontWeight: FontWeight.w800,
              color: AppTheme.inkBlack,
            ),
          ),
          for (final line in lines)
            Padding(
              padding: const EdgeInsets.only(top: 4),
              child: Text(line, style: AppTheme.matrixNumberStyle),
            ),
        ],
      ),
    );
  }
}
